package com.example.myapp;

public class Cart {
    private DigitalVideoDisc[] itemsOrdered;
    private int qtyOrdered = 0;
    private static final int MAX_NUMBERS_ORDERED = 20;

    // Phuong th?c thêm dia
    public void addDigitalVideoDisc(DigitalVideoDisc disc) {
        if (qtyOrdered < MAX_NUMBERS_ORDERED) {
            if (itemsOrdered == null) {
                itemsOrdered = new DigitalVideoDisc[MAX_NUMBERS_ORDERED];
            }
            itemsOrdered[qtyOrdered] = disc;
            qtyOrdered++;
            System.out.println("The disc has been added");
        } else {
            System.out.println("The cart is almost full");
        }
    }

    // Phuong th?c xóa dia
    public void removeDigitalVideoDisc(DigitalVideoDisc disc) {
        if (itemsOrdered != null && qtyOrdered > 0) {
            int indexToRemove = -1;
            for (int i = 0; i < qtyOrdered; i++) {
                if (itemsOrdered[i] != null && itemsOrdered[i].equals(disc)) {
                    indexToRemove = i;
                    break;
                }
            }
            if (indexToRemove != -1) {
                for (int i = indexToRemove; i < qtyOrdered - 1; i++) {
                    itemsOrdered[i] = itemsOrdered[i + 1];
                }
                itemsOrdered[qtyOrdered - 1] = null;
                qtyOrdered--;
                System.out.println("The disc has been removed");
            } else {
                System.out.println("The disc is not found in the cart");
            }
        } else {
            System.out.println("The cart is empty");
        }
    }

    // Phuong th?c tính t?ng chi phí
    public double totalCost() {
        double total = 0.0;
        if (itemsOrdered != null) {
            for (int i = 0; i < qtyOrdered; i++) {
                if (itemsOrdered[i] != null) {
                    total += itemsOrdered[i].getCost();
                }
            }
        }
        return total;
    }
    
    //Phuong thuc in thanh phan disc
    public void prinCart() {
    	for(int i =0;i<qtyOrdered ;i++) {
    		System.out.println( (i+1)+ "   " + itemsOrdered[i].getTitle() + "   " + itemsOrdered[i].getCost() );
    	}
    }
    
    // method overloading phuong thuc 1 ( them mang trong add )
    // => uu di?m là không ph?i t?o m?ng ch?a các dvd mà du?c hoàn toàn t? d?ng nhung th? cx là
    // nhu?c di?m t?i không di?m soát du?c, d? b? overhead nh?
 	public void addDigitalVideoDisc(DigitalVideoDisc[] dvdList) {
 	    for (DigitalVideoDisc dvd : dvdList) {
 	        if (qtyOrdered < MAX_NUMBERS_ORDERED) {
 	            itemsOrdered[qtyOrdered] = dvd;
 	            qtyOrdered++;
 	            System.out.println("The disc has been added: " + dvd.getTitle());
 	        } else {
 	            System.out.println("The cart is full. Cannot add " + dvd.getTitle());
 	            break;
 	        }
 	    }
 	}

Question: Is JAVA a Pass by Value or a Pass by Reference programming language?

Java luôn s? d?ng co ch? truy?n theo giá tr? (pass by value)

Code trong class TestPassingParameter : 

package com.example.myapp;

public class TestPassingParameter {
    public static void main(String[] args) {
        DigitalVideoDisc jungleDVD = new DigitalVideoDisc("Jungle", "Adventure", "Jon Favreau", 120, 20.0f);
        DigitalVideoDisc cinderellaDVD = new DigitalVideoDisc("Cinderella", "Fantasy", "Kenneth Branagh", 105, 18.5f);

        swap(jungleDVD, cinderellaDVD);  // Th? d?i ch? hai d?i tu?ng
        System.out.println("jungle DVD title: " + jungleDVD.getTitle());
        System.out.println("cinderella DVD title: " + cinderellaDVD.getTitle());

        changeTitle(jungleDVD, cinderellaDVD.getTitle());  // Ð?i tiêu d? c?a jungleDVD

        System.out.println("jungle DVD title: " + jungleDVD.getTitle());
    }

    public static void swap(DigitalVideoDisc dvd1, DigitalVideoDisc dvd2) {
        DigitalVideoDisc temp = dvd1;
        dvd1 = dvd2;
        dvd2 = temp;
    }

    public static void changeTitle(DigitalVideoDisc dvd, String title) {
        dvd.setTitle(title);
    }
}



1 ) After the call of swap(jungleDVD, cinderellaDVD) why does the title of these two 
objects still remain?

Ans : 
*  Java truy?n tham s? theo giá tr? (Pass by Value), k? c? khi tham s? là m?t d?i tu?ng.
*  Khi g?i swap(jungleDVD, cinderellaDVD), tham s? dvd1 và dvd2 trong phuong th?c ch? là b?n sao c?a các tham chi?u jungleDVD và cinderellaDVD trong main().
*  Vi?c gán l?i dvd1 = dvd2 và dvd2 = temp ch? thay d?i các b?n sao trong phuong th?c swap, không ?nh hu?ng d?n giá tr? ban d?u c?a jungleDVD và cinderellaDVD trong main().
2) After the call of changeTitle(jungleDVD, cinderellaDVD.getTitle()) why is 
the title of the JungleDVD changed?

Ans : 
*  Khi truy?n jungleDVD vào changeTitle(), Java truy?n m?t b?n sao c?a tham chi?u jungleDVD (t?c là b?n sao c?a d?a ch? ô nh? ch?a d?i tu?ng).
*  Khi th?c hi?n dvd.setTitle(title);, phuong th?c setTitle() tr?c ti?p thay d?i giá tr? c?a thu?c tính title bên trong d?i tu?ng mà tham chi?u dang tr? d?n.
*  Vì tham chi?u jungleDVD trong main() và dvd trong changeTitle() d?u tr? d?n cùng m?t d?i tu?ng trong b? nh?, nên s? thay d?i trong phuong th?c cung ?nh hu?ng d?n d?i tu?ng g?c.
=> JAVA is always a pass by value programming language.
